// HIGH-SENSITIVITY CARTON ANIMATION let cartonTarget = { x: 0, y: 0, z: 0 }; let cartonMotion = { x: 0, y: 0, z: 0 }; let cartonFrameStarted = false; function updateCartonTarget(x, y, z) { // Increased sensitivity: +/-4G visual range const visualRangeG = 4; const clamp = (value) => Math.max(-1, Math.min(1, value)); cartonTarget.x = clamp(x / visualRangeG); cartonTarget.y = clamp(y / visualRangeG); cartonTarget.z = clamp(z / visualRangeG); document.getElementById("boxX").textContent = x.toFixed(2); document.getElementById("boxY").textContent = y.toFixed(2); document.getElementById("boxZ").textContent = z.toFixed(2); if (!cartonFrameStarted) { cartonFrameStarted = true; requestAnimationFrame(animateCarton); } } function animateCarton() { // Faster response to sensor movement const response = 0.28; cartonMotion.x += (cartonTarget.x - cartonMotion.x) * response; cartonMotion.y += (cartonTarget.y - cartonMotion.y) * response; cartonMotion.z += (cartonTarget.z - cartonMotion.z) * response; const carton = document.getElementById("carton"); const shadow = document.getElementById("boxShadow"); if (carton) { const moveX = cartonMotion.x * 58; const moveY = cartonMotion.y * -48; const moveZ = cartonMotion.z * 22; const rotateX = cartonMotion.y * 48; const rotateY = cartonMotion.x * 52; const rotateZ = cartonMotion.z * 34; const scale = 1 + cartonMotion.z * 0.12; carton.style.transform = `translate3d(${moveX}px, ${moveY}px, ${moveZ}px) ` + `rotateX(${rotateX}deg) rotateY(${rotateY}deg) ` + `rotateZ(${rotateZ}deg) scale(${scale})`; } if (shadow) { shadow.style.transform = `translate(${cartonMotion.x * 22}px, ${69 + cartonMotion.y * 10}px) ` + `scale(${1 - cartonMotion.z * 0.10})`; } if ( Math.abs(cartonTarget.x - cartonMotion.x) > 0.001 || Math.abs(cartonTarget.y - cartonMotion.y) > 0.001 || Math.abs(cartonTarget.z - cartonMotion.z) > 0.001 ) { requestAnimationFrame(animateCarton); } else { cartonFrameStarted = false; } }
